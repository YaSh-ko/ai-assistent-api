62 @staticmethod
63 def hash_password(password: str) -> str:
64 """Hash password using SHA256 with salt."""
65 salt = secrets.token_hex(16)
66 password_hash = hashlib.sha256((password + salt).encode()).hexdigest()
67 return f"{salt}:{password_hash}"
68
69 @staticmethod
70 def verify_password(password: str, stored_hash: str) -> bool:
71 """Verify password against stored hash."""
72 if not password or not stored_hash:
73 return False
74 try:
75 if ":" not in stored_hash:
76 return False
77 salt, password_hash = stored_hash.split(":", 1)
78 if not salt or not password_hash:
79 return False
80 return hashlib.sha256((password + salt).encode()).hexdigest() == password_hash
81 except (ValueError, AttributeError):
82 return False

247 async def validate_session(self, token: str) -> Tuple[User, Session]:
248 """Validate session: JWT RS256 или устаревший opaque-токен в БД."""
249 if looks_like_jwt(token):
250 if not settings.JWT_PUBLIC_KEY_PEM:
251 raise UnauthorizedException("Invalid session token")
252 return await self._validate_session_rs256(token)
253 return await self._validate_session_opaque(token)

255 async def _validate_session_rs256(self, token: str) -> Tuple[User, Session]:
256 now = datetime.now(timezone.utc)
257 try:
258 claims = decode_session_jwt(
259 settings.JWT_PUBLIC_KEY_PEM,
260 token,
261 issuer=settings.JWT_ISSUER or None,
262 audience=settings.JWT_AUDIENCE or None,
263 leeway_seconds=settings.JWT_LEEWAY_SECONDS,
264 )
265 except pyjwt.PyJWTError as e:
266 logger.debug("JWT decode failed: %s", e)
267 raise UnauthorizedException() from e
268
269 session_id = claims.get("sid")
270 user_id = claims.get("sub")
271 if not session_id or not user_id:
272 raise UnauthorizedException()
273
274 session = await self.get_session_by_id(session_id)
275 if not session or session.user_id != user_id:
276 raise UnauthorizedException()
277 if not tokens_equal_constant_time(session.token, token):
278 raise UnauthorizedException()
279
280 if session.expires_at < now:
281 await self.db.execute(delete(Session).where(Session.id == session.id))
282 await self.db.commit()
283 raise SessionExpiredException()
284
285 user = await self.get_user_by_id(user_id)
286 if not user:
287 raise UnauthorizedException()
288 return user, session

311 async def delete_session(self, token: str) -> bool:
312 """Delete session by token (полное совпадение строки в БД или JWT + сверка по sid)."""
313 session = await self.get_session_by_token(token)
314 if session:
315 await self.db.execute(
316 delete(Session).where(Session.id == session.id)
317 )
318 await self.db.commit()
319 return True
320 if looks_like_jwt(token) and settings.JWT_PUBLIC_KEY_PEM:
321 try:
322 claims = decode_session_jwt(
323 settings.JWT_PUBLIC_KEY_PEM,
324 token,
325 issuer=settings.JWT_ISSUER or None,
326 audience=settings.JWT_AUDIENCE or None,
327 leeway_seconds=settings.JWT_LEEWAY_SECONDS,
328 )
329 except pyjwt.PyJWTError:
330 return False
331 sid = claims.get("sid")
332 if isinstance(sid, str) and sid.strip():
333 row = await self.get_session_by_id(sid)
334 if row and tokens_equal_constant_time(row.token, token):
335 await self.db.execute(delete(Session).where(Session.id == row.id))
336 await self.db.commit()
337 return True
338 return False
