32@router.get("", response_model=GoalListResponse)
33async def get_goals(
34 user_id: Annotated[str, Depends(get_current_user_id)],
35):
36 """Get list of goals from Neo4j."""
37 query = """
38 MATCH (g:Goal {user_id: $user_id})
39 RETURN g
40 ORDER BY g.created_at DESC
41 """
42 results = await neo4j_client.execute_query_async(query, {"user_id": user_id})
43 goals = [GoalResponse(**record["g"]) for record in results]
44 return GoalListResponse(goals=goals)
