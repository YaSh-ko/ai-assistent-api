79@router.get("/thread/{thread_id}/context", response_model=ThreadContextResponse)
80async def get_thread_context(
81 thread_id: str,
82 user_id: Annotated[str, Depends(get_current_user_id)],
83 db: Annotated[AsyncSession, Depends(get_db)],
84):
85 """Get context for a conversation thread (e.g., related Entry)."""
86 stmt = select(EntryThread).where(
87 EntryThread.thread_id == thread_id,
88 EntryThread.user_id == user_id
89 )
90 result = await db.execute(stmt)
91 entry_thread = result.scalar_one_or_none()
92
93 if not entry_thread:
94 return ThreadContextResponse(
95 type="general",
96 title="Свободный диалог",
97 description="Ни к чему не привязанный чатик",
98 entry_id=None
99 )
100
101 stmt_entry = select(Entry).where(Entry.id == entry_thread.entry_id)
102 entry_result = await db.execute(stmt_entry)
103 entry = entry_result.scalar_one_or_none()
104
105 if not entry:
106 return ThreadContextResponse(
107 type="general",
108 title="Свободный диалог",
109 description="Ни к чему не привязанный чатик",
110 entry_id=None
111 )
112
113 # Форматируем короткое описание
114 desc = entry.description or ""
115 if len(desc) > 100:
116 desc = desc[:100] + "..."
117
118 return ThreadContextResponse(
119 type="event",
120 title=entry.title or "Событие",
121 description=desc,
122 entry_id=entry.id
