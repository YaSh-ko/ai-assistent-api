122 def parse_cors_origins(cls, v: str | list) -> list[str]:
123 """Парсит CORS_ORIGINS из различных форматов."""
124 if isinstance(v, list):
125 return v
126 if isinstance(v, str):
127 # Пробуем распарсить как JSON
128 try:
129 parsed = json.loads(v)
130 if isinstance(parsed, list):
131 return parsed
132 except json.JSONDecodeError:
133 pass
134 # Если не JSON, пробуем разделить по запятой
135 if "," in v:
136 return [origin.strip() for origin in v.split(",") if origin.strip()]
137 # Иначе возвращаем как список с одним элементом
138 return [v.strip()] if v.strip() else []
