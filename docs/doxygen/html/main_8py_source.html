64def _ensure_cors_origins_list() -> None:
65 """Приводит CORS_ORIGINS к списку и логирует."""
66 logger.info("CORS origins configured: %s", settings.CORS_ORIGINS)
67 if isinstance(settings.CORS_ORIGINS, list):
68 return

76def _add_cors_to_response(response: JSONResponse, request: Request) -> JSONResponse:
77 """Добавляет CORS-заголовки к ответу при разрешённом origin."""
78 origin = request.headers.get("Origin")
79 if origin and origin in settings.CORS_ORIGINS:
80 response.headers["Access-Control-Allow-Origin"] = origin
81 response.headers["Access-Control-Allow-Credentials"] = "true"
82 response.headers["Access-Control-Allow-Methods"] = "GET, POST, PUT, DELETE, OPTIONS, PATCH"
83 response.headers["Access-Control-Allow-Headers"] = "Content-Type, Authorization, Accept, Origin, X-Requested-With"
84 return response

106def _register_exception_handlers(app: FastAPI) -> None:
107 """Регистрирует глобальные обработчики исключений."""

109 @app.exception_handler(RequestValidationError)
110 async def _validation_handler(request: Request, exc: RequestValidationError):
111 r = JSONResponse(
112 status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
113 content={"error": "Validation Error", "message": "Invalid request data", "details": exc.errors()},
114 )
115 return _add_cors_to_response(r, request)
116
117 @app.exception_handler(SQLAlchemyError)
118 async def _db_handler(request: Request, exc: SQLAlchemyError):
119 logger.error("Database error: %s", exc, exc_info=True)
120 err_str = str(exc).lower()
121 msg = "An error occurred while processing your request"
122 if settings.DEBUG:
123 msg = f"Database error: {exc}"
124 if "connection" in err_str or "connect" in err_str:
125 msg = "Database connection error. Please check database configuration."
126 elif "constraint" in err_str or "unique" in err_str:
127 msg = "Database constraint violation" if not settings.DEBUG else f"Database constraint error: {exc}"
128 r = JSONResponse(
129 status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
130 content={"error": "Database Error", "message": msg, "type": exc.__class__.__name__ if settings.DEBUG else None},
131 )
132 return _add_cors_to_response(r, request)
133
134 @app.exception_handler(InvalidCredentialsException)
135 @app.exception_handler(UserAlreadyExistsException)
136 @app.exception_handler(InvalidTokenException)
137 @app.exception_handler(UnauthorizedException)
138 @app.exception_handler(BadRequestException)
139 @app.exception_handler(JwtConfigurationException)
140 async def _auth_handler(request: Request, exc: Exception):
141 code = getattr(exc, "status_code", status.HTTP_400_BAD_REQUEST)
142 detail = getattr(exc, "detail", str(exc))
143 r = JSONResponse(status_code=code, content={"error": exc.__class__.__name__, "message": detail})
144 return _add_cors_to_response(r, request)
145
146 @app.exception_handler(Exception)
147 async def _general_handler(request: Request, exc: Exception):
148 logger.error("Unhandled exception: %s", exc, exc_info=True)
149 msg = "An unexpected error occurred" if not settings.DEBUG else str(exc)
150 r = JSONResponse(status_code=status.HTTP_500_INTERNAL_SERVER_ERROR, content={"error": "Internal Server Error", "message": msg})
151 return _add_cors_to_response(r, request)
