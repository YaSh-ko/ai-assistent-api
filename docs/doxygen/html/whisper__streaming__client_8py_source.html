51 def _transcribe_file_sync(
52 self,
53 audio_file: BinaryIO,
54 language: Optional[str],
55 kwargs: dict,
56 ) -> dict:
57 """Blocking transcription (run in executor)."""
58 if self.model is None:
59 raise RuntimeError("Whisper model not loaded")
60 with tempfile.NamedTemporaryFile(delete=False, suffix=".wav") as tmp_file:
61 tmp_path = tmp_file.name
62 audio_file.seek(0)
63 tmp_file.write(audio_file.read())
64 try:
65 result = self.model.transcribe(
66 tmp_path,
67 language=language or self.language,
68 **kwargs
69 )
70 return {
71 "text": result.get("text", "").strip(),
72 "language": result.get("language", "unknown"),
73 "segments": result.get("segments", []),
74 "duration": self._calculate_duration(result.get("segments", [])),
75 }
76 finally:
77 if os.path.exists(tmp_path):
78 os.unlink(tmp_path)

133 async def transcribe_stream(
134 self,
135 audio_stream: AsyncGenerator[bytes, None],
136 language: Optional[str] = None,
137 chunk_size: int = 16000, # ~1 second of audio at 16kHz
138 **kwargs
139 ) -> AsyncGenerator[dict, None]:
140 """
141 Transcribe audio stream (streaming mode).
142
143 Note: Whisper doesn't natively support streaming, so we process chunks
144 and yield partial results. For true streaming, consider whisper_streaming library.
145 """
146 audio_chunks = []
147 segment_index = 0
148
149 async for chunk in audio_stream:
150 audio_chunks.append(chunk)
151 total = len(b"".join(audio_chunks))
152 if total < chunk_size:
153 continue
154
155 out = self._transcribe_chunk(
156 b"".join(audio_chunks),
157 language,
158 segment_index,
159 is_final=False,
160 **kwargs,
161 )
162 if out:
163 yield out
164 segment_index += 1
165 audio_chunks = []
166
167 if not audio_chunks:
168 return
169 out = self._transcribe_chunk(
170 b"".join(audio_chunks),
171 language,
172 segment_index,
173 is_final=True,
174 **kwargs,
175 )
176 if out:
177 yield out
