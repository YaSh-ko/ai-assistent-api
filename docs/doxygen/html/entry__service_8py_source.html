25async def sync_entry_to_neo4j(
26 entry: Entry,
27 language: str = "",
28 duration: float = 0.0,
29) -> None:
30 """Sync Entry to Neo4j graph. Does not raise on failure."""
31 try:
32 query = """
33 MERGE (e:Entry {id: $entry_id})
34 SET e.user_id = $user_id,
35 e.content = $content,
36 e.timestamp = datetime($timestamp),
37 e.word_count = $word_count,
38 e.transcription_language = $language,
39 e.audio_duration = $duration,
40 e.updated_at = datetime()
41 RETURN e
42 """
43 word_count = len((entry.description or "").split())
44 timestamp = datetime.combine(entry.event_date, datetime.min.time())
45 await neo4j_client.execute_query_async(
46 query,
47 {
48 "entry_id": str(entry.id),
49 "user_id": entry.user_id,
50 "content": entry.description or "",
51 "timestamp": timestamp.isoformat(),
52 "word_count": word_count,
53 "language": language,
54 "duration": duration,
55 },
56 )
57 logger.info("Entry %s synced to Neo4j", entry.id)
58 except Exception as e:
59 logger.error("Error syncing entry to Neo4j: %s", e)
