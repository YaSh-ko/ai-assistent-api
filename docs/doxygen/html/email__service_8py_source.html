37 async def send_email(
38 self,
39 to_email: str,
40 subject: str,
41 html_content: str,
42 text_content: Optional[str] = None
43 ) -> bool:
44 """
45 Send an email via SMTP.
46
47 Args:
48 to_email: Recipient email address
49 subject: Email subject
50 html_content: HTML content of the email
51 text_content: Optional plain text fallback
52
53 Returns:
54 True if email was sent successfully, False otherwise
55 """
56 if not self.smtp_username or not self.smtp_password:
57 logger.warning("SMTP credentials not configured, skipping email send")
58 return False
59
60 # Debug logging
61 logger.info(f"Attempting to send email to {to_email}")
62 logger.info(f"SMTP config: host={self.smtp_host}, port={self.smtp_port}, from={self.from_email}")
63
64 try:
65 # Create message
66 message = MIMEMultipart("alternative")
67 # Use formataddr with Header for proper RFC 2047 encoding
68 message["From"] = formataddr((str(Header(self.from_name, "utf-8")), self.from_email))
69 message["To"] = to_email
70 message["Subject"] = str(Header(subject, "utf-8"))
71
72 # Add text part if provided
73 if text_content:
74 text_part = MIMEText(text_content, "plain", "utf-8")
75 message.attach(text_part)
76
77 # Add HTML part
78 html_part = MIMEText(html_content, "html", "utf-8")
79 message.attach(html_part)
80
81 # Send email via SMTP
82 # Port 465 usually uses implicit TLS (SSL), while 587 uses STARTTLS
83 use_tls = self.smtp_port == 465
84 start_tls = self.smtp_port != 465
85
86 logger.info(f"Connecting to SMTP: use_tls={use_tls}, start_tls={start_tls}")
87
88 await aiosmtplib.send(
89 message,
90 hostname=self.smtp_host,
91 port=self.smtp_port,
92 username=self.smtp_username,
93 password=self.smtp_password,
94 use_tls=use_tls,
95 start_tls=start_tls,
96 )
97
98 logger.info(f"Email sent successfully to {to_email}")
99 return True
100
101 except aiosmtplib.SMTPException as e:
102 logger.error(f"SMTP error sending email to {to_email}: {str(e)}")
103 return False
104 except Exception as e:
105 logger.error(f"Error sending email to {to_email}: {str(e)}")
106 return False
