1"""
2@file jwt_auth.py
3@brief JWT-токены сессий на алгоритме RS256 (RSA + SHA-256).
4
5Модуль отвечает за выпуск и проверку подписанных токенов. Намеренно
6ограничен только RS256, чтобы исключить «algorithm confusion» атаки.
7Входная строка токена проверяется на максимальную длину до декодирования —
8защита от DoS через огромные payload'ы.

11- Явный список алгоритмов (только RS256).
12- Ограничение длины входной строки до декодирования.
13- Сравнение токенов через secrets.compare_digest (constant-time).

36def normalize_pem(pem: str) -> str:
37 """
38 Приводит PEM из .env к нормальному виду (часто передают \\n вместо перевода строки).
39 Удаляет случайные кавычки и слэши из переменных окружения GitLab CI.
40 """
41 s = pem.strip()
42
43 # В валидном PEM (Base64 + заголовки) никогда не бывает кавычек
44 s = s.replace('"', '').replace("'", "")
45
46 # Обрабатываем возможные экранирования переносов строк (\\n или \n)
47 s = s.replace("\\\\n", "\n")
48 s = s.replace("\\n", "\n")
49 s = s.replace("\\\\r", "")
50 s = s.replace("\\r", "")
51
52 # Удаляем оставшиеся висячие бэк-слэши (например, от экранированной кавычки \")
53 s = s.replace("\\", "")
54
55 return s.strip()

58def looks_like_jwt(token: str) -> bool:
59 """
60 Эвристика: три сегмента base64url через точку.
61 Не гарантирует валидный JWT, но отсекает явный мусор.
62 """
63 if not token or len(token) > MAX_JWT_STRING_LENGTH:
64 return False
65 if token.count(".") != 2:
66 return False
67 # base64url: A-Za-z0-9-_
68 if not re.match(r"^[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+$", token):
69 return False
70 return True

105def decode_session_jwt(
106 public_key_pem: str,
107 token: str,
108 *,
109 issuer: Optional[str] = None,
110 audience: Optional[str] = None,
111 leeway_seconds: int = 0,
112) -> dict[str, Any]:
113 """
114 Проверяет подпись RS256 и возвращает payload.
115
116 leeway_seconds — допуск по часам между клиентом и сервером (nbf/exp).
117 """
118 if len(token) > MAX_JWT_STRING_LENGTH:
119 raise jwt.InvalidTokenError("Token too long")
120
121 require = ["exp", "iat", "sub", "sid"]
122 if issuer:
123 require.append("iss")
124 if audience:
125 require.append("aud")
126
127 kwargs: dict[str, Any] = {
128 "algorithms": [JWT_ALGORITHM],
129 "options": {
130 "require": require,
131 "verify_signature": True,
132 },
133 "leeway": leeway_seconds,
134 }
135 if issuer:
136 kwargs["issuer"] = issuer
137 if audience:
138 kwargs["audience"] = audience
139
140 claims = jwt.decode(token, normalize_pem(public_key_pem), **kwargs)
141
142 tu = claims.get(CLAIM_TOKEN_USE)
143 if tu is not None and tu != TOKEN_USE_SESSION:
144 raise jwt.InvalidTokenError("Invalid token_use")
145
146 sub = claims.get("sub")
147 sid = claims.get("sid")
148 if not isinstance(sub, str) or not isinstance(sid, str) or not sub.strip() or not sid.strip():
149 raise jwt.InvalidTokenError("Invalid sub/sid")
150
151 return claims
