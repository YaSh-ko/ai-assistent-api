91@router.get("/{id}/analysis", response_model=EntryAnalysisResponse)
92async def get_entry_analysis(
93 id: UUID,
94 user_id: Annotated[str, Depends(get_current_user_id)],
95 db: Annotated[AsyncSession, Depends(get_db)],
96):
97 """Get complete analysis for an entry."""
98 entry_repo = EntryRepository(db)
99 entry = await entry_repo.get_by_id(id)
100
101 if not entry or entry.user_id != user_id:
102 raise HTTPException(
103 status_code=status.HTTP_404_NOT_FOUND,
104 detail=MSG_ENTRY_NOT_FOUND
105 )
106
107 # Get intensity metrics
108 metrics_repo = IntensityMetricRepository(db)
109 metrics = await metrics_repo.get_by_entity("entry", id)
110
111 # Get related situations
112 related_repo = RelatedSituationRepository(db)
113 related = await related_repo.get_by_source("entry", id)
114
115 # Get negative impacts
116 impact_repo = NegativeImpactRepository(db)
117 impacts = await impact_repo.get_by_source("entry", id)
118
119 # Get transformations
120 trans_repo = TransformationRepository(db)
121 transformations = await trans_repo.get_by_source("entry", id)
122
123 # Get concepts from Neo4j
124 concepts = await neo4j_client.get_related_nodes(str(id), "MENTIONS", "Concept")
125
126 return EntryAnalysisResponse(
127 entry=EntryResponse.model_validate(entry),
128 intensity_metrics=[IntensityMetricResponse.model_validate(m) for m in metrics],
129 related_situations=[RelatedSituationResponse.model_validate(r) for r in related],
130 negative_impacts=[NegativeImpactResponse.model_validate(i) for i in impacts],
131 transformations=[TransformationResponse.model_validate(t) for t in transformations],
132 concepts=concepts
133 )
