30def _rhizome_node_labels(node_types: Optional[List[str]]) -> List[str]:
31 """Return list of node labels to query, optionally filtered by node_types."""
32 if not node_types:

37def _rhizome_time_filter(time_period: Optional[str]) -> str:
38 """Return Cypher snippet for time filter on Entry nodes, or empty string."""
39 if not time_period:
40 return ""
41 if time_period == "past":
42 return "AND n.timestamp < datetime()"
43 if time_period == "present":
44 return "AND n.timestamp >= datetime() - duration({days: 30}) AND n.timestamp <= datetime()"
45 if time_period == "future":
46 return "AND n.timestamp > datetime()"
47 return ""

50def _node_description(node: Dict[str, Any], node_type: Optional[str]) -> str:
51 """Build short description for a node by its type."""
52 if node_type == "Entry":
53 return node.get("content_summary", node.get("content", ""))[:100]
54 if node_type == "Concept":
55 return node.get("description", node.get("name", ""))[:100]
56 if node_type == "Goal":
57 return node.get("description", node.get("title", ""))[:100]
58 if node_type == "Experiment":
59 return node.get("description", node.get("title", ""))[:100]
60 if node_type == "Analysis":
61 return node.get("summary", node.get("content", ""))[:100]
62 return ""

65def _node_to_graph_item(node: Dict[str, Any], node_id: str, user_id: str, node_type: Optional[str]) -> Dict[str, Any]:
66 """Build a single node dict for graph response."""
67 description = _node_description(node, node_type)
68
69 props = {}
70 for k, v in node.items():
71 if k in ["id", "user_id"]:
72 continue
73 # Serialize database-specific date/time objects for Pydantic
74 if hasattr(v, "iso_format"):
75 props[k] = v.iso_format()
76 elif hasattr(v, "isoformat"):
77 props[k] = v.isoformat()
78 else:
79 props[k] = v
80
81 return {
82 "id": node_id,
83 "user": user_id,
84 "description": description,
85 "type": node_type,
86 **props
87 }

125 def execute_query(self, query: str, parameters: Optional[Dict[str, Any]] = None) -> List[Dict[str, Any]]:
126 """Execute a Cypher query and return results."""
127 self._ensure_connected()
128
129 if not self.driver:
130 raise RuntimeError("Neo4j driver not initialized")
131
132 try:
133 with self.driver.session() as session:
134 result = session.run(cast(LiteralString, query), parameters or {})
135 return [record.data() for record in result]
136 except Exception as e:
137 logger.error(f"Neo4j query error: {e}")
138 raise

140 async def execute_query_async(self, query: str, parameters: Optional[Dict[str, Any]] = None) -> List[Dict[str, Any]]:
141 """Execute a Cypher query asynchronously and return results."""
142 loop = asyncio.get_event_loop()
143 return await loop.run_in_executor(None, partial(self.execute_query, query, parameters))

145 async def get_rhizome_graph(
146 self,
147 user_id: str,
148 node_types: Optional[List[str]] = None,
149 time_period: Optional[str] = None
150 ) -> Dict[str, Any]:
151 """
152 Get the complete graph for a user in format suitable for react-force-graph.
153 """
154 node_labels = _rhizome_node_labels(node_types)
155
156 # Строим условия выборки через OR, чтобы поддерживать серверы с Neo4j 4.x
157 if node_labels:
158 label_cond_n = " OR ".join([f"n:{lbl}" for lbl in node_labels])
159 label_cond_m = " OR ".join([f"m:{lbl}" for lbl in node_labels])
160 else:
161 label_cond_n = "FALSE"
162 label_cond_m = "FALSE"
163
164 time_filter = _rhizome_time_filter(time_period)
165
166 query = f"""
167 MATCH (n)
168 WHERE ({label_cond_n}) AND n.user_id = $user_id {time_filter}
169 OPTIONAL MATCH (n)-[r]->(m)
170 WITH n, labels(n) as n_labels,
171 collect(CASE WHEN r IS NOT NULL AND m IS NOT NULL THEN {{
172 target: CASE WHEN m.id IS NOT NULL THEN toString(m.id) ELSE coalesce(m.name, m.title, toString(id(m))) END,
173 type: type(r),
174 m_labels: labels(m),
175 m_node: m,
176 rel_props: properties(r)
177 }} ELSE null END) as raw_rels
178 WITH n, n_labels,
179 [x IN raw_rels WHERE x IS NOT NULL] as relationships
180 RETURN n, n_labels, relationships
181 """
182
183 results = await self.execute_query_async(query, {"user_id": user_id})
184 nodes = []
185 links = []
186 node_ids = set()
187
188 for record in results:
189 node = record["n"]
190 n_labels = record.get("n_labels", [])
191 node_id = str(node.get("id", ""))
192 if node_id and node_id not in node_ids:
193 node_ids.add(node_id)
194 node_type = next((lbl for lbl in node_labels if lbl in n_labels), None)
195 nodes.append(_node_to_graph_item(node, node_id, user_id, node_type))
196 relationships = record.get("relationships", [])
197 for rel in relationships:
198 if not rel or not rel.get("target"):
199 continue
200 target_id = str(rel["target"])
201 rel_props = rel.get("rel_props") or {}
202 links.append({
203 "source": node_id,
204 "target": target_id,
205 "type": rel.get("type", "RELATED"),
206 "reason": rel_props.get("reason") or rel_props.get("justification") or rel_props.get("description") or "",
207 })
208 # Add target node if not already present (e.g. Topic nodes)
209 if target_id not in node_ids and rel.get("m_node"):
210 m_node = rel["m_node"]
211 m_labels = rel.get("m_labels", [])
212 m_type = next((lbl for lbl in RHIZOME_NODE_LABELS if lbl in m_labels), m_labels[0] if m_labels else "Unknown")
213 node_ids.add(target_id)
214 nodes.append(_node_to_graph_item(dict(m_node), target_id, user_id, m_type))
215
216 logger.info(
217 "get_rhizome_graph: user=%s nodes=%d links=%d sample_rels=%s",
218 user_id, len(nodes), len(links),
219 [r.get("relationships") for r in results[:3]] if results else []
220 )
221 return {"nodes": nodes, "links": links}

223 async def search_nodes(self, user_id: str, query: str, limit: int = 50) -> List[Dict[str, Any]]:
224 """Search nodes by name or title."""
225 search_query = """
226 MATCH (n)
227 WHERE n.user_id = $user_id
228 AND (
229 (n:Entry AND (n.content CONTAINS $query OR n.content_summary CONTAINS $query))
230 OR (n:Concept AND n.name CONTAINS $query)
231 OR (n:Goal AND n.title CONTAINS $query)
232 OR (n:Experiment AND n.title CONTAINS $query)
233 OR (n:Analysis AND (n.title CONTAINS $query OR n.content CONTAINS $query))
234 )
235 RETURN n, labels(n) as n_labels
236 LIMIT $limit
237 """
238
239 results = await self.execute_query_async(search_query, {
240 "user_id": user_id,
241 "query": query,
242 "limit": limit
243 })
244
245 nodes = []
246 for record in results:
247 node = record["n"]
248 n_labels = record.get("n_labels", [])
249 node_id = str(node.get("id", ""))
250 node_type = next((lbl for lbl in RHIZOME_NODE_LABELS if lbl in n_labels), None)
251 nodes.append(_node_to_graph_item(node, node_id, user_id, node_type))
252 return nodes
